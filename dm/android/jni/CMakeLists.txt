# The Android native build: the host (dm/host + dm/gfx C sources), the POSIX/Mach shim layer the host's
# Apple includes compile against (shim/), the Android front end and JNI bridge, and the Vulkan
# execution layer (nvn/nvn_vk.c) - one shared library, libdmhost.so, loaded by the app
# (System.loadLibrary("dmhost"); HostJni's native methods are in dm_jni.c).
#
# The host's source list comes from dm/host/Makefile itself (tools/gen_build_android.py reads the
# expanded compile commands and writes host_sources.cmake): GAME=<profile id> picks the game, exactly
# as the Mac build's make does.
cmake_minimum_required(VERSION 3.22.1)
project(dmhost C ASM)

get_filename_component(DM ${CMAKE_CURRENT_SOURCE_DIR}/../.. ABSOLUTE)   # the dm/ directory (dm/android/jni → dm): the host, gfx and games paths hang off it

# The host's sources, includes and the game profile header, generated from the Makefile (regenerated
# when it changes; a stale list stops the build the same way the Mac build does when GAME is not set)
if(NOT DEFINED ENV{GAME})
  execute_process(COMMAND python3 ${DM}/games/game.py list OUTPUT_VARIABLE list OUTPUT_STRIP_TRAILING_WHITESPACE)
  message(FATAL_ERROR "GAME is not set: cmake -DGAME=<base game title ID> .. (profiles: ${list})")
endif()
execute_process(
  COMMAND python3 ${DM}/android/tools/gen_build_android.py ${DM}/host ${CMAKE_CURRENT_BINARY_DIR}/host_sources.cmake
  WORKING_DIRECTORY ${DM}/host
  RESULT_VARIABLE gen_res
  OUTPUT_VARIABLE gen_out ERROR_VARIABLE gen_err)
if(NOT gen_res EQUAL 0)
  message(FATAL_ERROR "gen_build_android.py failed res=[${gen_res}] (GAME=$ENV{GAME}): err=[${gen_err}] out=[${gen_out}]")
endif()
include(${CMAKE_CURRENT_BINARY_DIR}/host_sources.cmake)

# The counting trampolines for ELF (gen_protos.py --elf: no underscore on a C symbol's assembly name,
# adrp + :lo12: addressing; the committed Mach-O nvn_thunks.S is untouched)
execute_process(
  COMMAND python3 ${DM}/gfx/tools/gen_protos.py ${DM}/gfx/nvn/include/nvn.h ${DM}/gfx/nvn/names.txt
          ${CMAKE_CURRENT_BINARY_DIR}/nvn_protos.h --elf ${CMAKE_CURRENT_BINARY_DIR}/nvn_thunks_elf.S
  WORKING_DIRECTORY ${DM} RESULT_VARIABLE proto_res)
if(NOT proto_res EQUAL 0)
  message(FATAL_ERROR "gen_protos.py --elf failed")
endif()

# nvn_protos.h (the signature-check table hle_nvn.c includes) is the committed one in dm/gfx/nvn -
# gen_protos.py writes the same content; only the trampolines differ between the object formats
set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/nvn_thunks_elf.S PROPERTIES LANGUAGE ASM)

add_library(dmhost SHARED
  ${DM_HOST_SOURCES}
  ${DM_HOST_MAIN}
  ${CMAKE_CURRENT_BINARY_DIR}/nvn_thunks_elf.S
  ${CMAKE_CURRENT_SOURCE_DIR}/sys_shim.c
  ${CMAKE_CURRENT_SOURCE_DIR}/mach_time_outofline.c   # the symbol nvn_entry.S links against (the header's is inline)
  ${CMAKE_CURRENT_SOURCE_DIR}/audio_shim.c
  ${CMAKE_CURRENT_SOURCE_DIR}/video_shim.c
  ${CMAKE_CURRENT_SOURCE_DIR}/front_android.c
  ${CMAKE_CURRENT_SOURCE_DIR}/dm_jni.c
  ${CMAKE_CURRENT_SOURCE_DIR}/haptics_android.c
  ${CMAKE_CURRENT_SOURCE_DIR}/memstat_android.c
  ${CMAKE_CURRENT_SOURCE_DIR}/nvn/nvn_vk.c)

# The game profile header is generated by the Makefile's rule (gen_build_android.py ran it); the ELF
# thunks' directory carries the generated nvn_protos.h beside the sources that include it
set_property(SOURCE ${DM_HOST_MAIN} APPEND PROPERTY COMPILE_DEFINITIONS main=dm_main)

# Include order: the shim directory first, so the host's Apple includes (<mach/mach.h>, <AudioToolbox/…>)
# resolve to the shims; then the host's own include paths from the Makefile, the generated headers'
# directory, and the Vulkan headers the repository ships (the NDK has them too; the repo's are the
# tested ones, dm/third_party/README.md)
target_include_directories(dmhost PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}/shim
  ${DM}/host   # dm.h, hle.h, front.h: the Android front end's sources include them like the host's own do
  ${DM}/android/jni
  ${DM}/android/jni/nvn
  ${DM}/gfx/nvn   # nvn_mtl.h / nvn_obj.h: the Vulkan layer implements the same seam the Metal one does
  ${DM_HOST_INCLUDES}
  ${CMAKE_CURRENT_BINARY_DIR}
  ${DM}/third_party/vulkan-headers/include)

# The guest code is position-independent by construction (the rewriter makes it so); the host's
# .S files need the LSE atomics the counting thunks use (ARMv8.1, every Android device since the
# Pixel 2's Snapdragon 835; the Makefile's comment about A12 / M1 is the Apple statement of it)
target_compile_options(dmhost PRIVATE -Wall -Wno-unused-function -O1 -g -fno-strict-aliasing -march=armv8.1-a)
target_compile_definitions(dmhost PRIVATE DM_ANDROID=1)
# LTO off: the host's sections (swhle / swforce / swnvn) must survive as their own sections, and
# the section walkers find them by name
set_target_properties(dmhost PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)

# Android's native media: AAudio (the output unit), NdkMediaCodec (movie decoding), and the runtime
# Vulkan loader (dlopen'd by nvn_vk.c: libvulkan.so need not be linked, a device without it still boots)
# -Bsymbolic: the host's assembly (the gate, the NVN thunks) takes symbol addresses with adrp/add,
# which cannot go through the PLT a preemptible shared-library symbol would need. Everything the
# .so defines is resolved inside it - nothing outside overrides the host's own symbols
target_link_options(dmhost PRIVATE -Wl,-Bsymbolic)
target_link_libraries(dmhost PRIVATE log android aaudio mediandk)
